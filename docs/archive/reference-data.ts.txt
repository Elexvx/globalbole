export const getCategory = (slug: string) => categories.find((category) => category.slug === slug);

export const getStaticStory = (slug: string) => stories.find((story) => story.slug === slug);

export const getStoriesByCategory = (category: CategorySlug) =>
  stories.filter((story) => story.category === category);

export const sortStories = (items: Story[]) =>
  [...items].sort((a, b) => b.date.localeCompare(a.date));
